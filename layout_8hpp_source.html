145inline auto flex_share(flex_policy policy, double extra, std::size_t fill_count, double finite_grow)
146 -> double {
147 if (extra > 0.0 && fill_count > 0) {
148 return policy.fill ? extra / static_cast<double>(fill_count) : 0.0;
149 }
150 if (extra < 0.0 && finite_grow > 0.0) {
151 return extra * policy.grow / finite_grow;
152 }
153 if (extra < 0.0 && fill_count > 0) {
154 return policy.fill ? extra / static_cast<double>(fill_count) : 0.0;
155 }
156 if (finite_grow > 0.0) {
157 return extra * policy.grow / finite_grow;
158 }
159 return 0.0;
160}

162template <stack_axis Axis>
163inline void distribute_flex(layout_result &result, std::span<flex_policy const> flex,
164 double extra) {
165 std::size_t fill_count = 0;
166 double finite_grow = 0.0;
167
168 for (auto policy : flex) {
169 if (policy.fill) {
170 ++fill_count;
171 } else {
172 finite_grow += policy.grow;
173 }
174 }
175
176 for (std::size_t i = 0; i < result.children.size(); ++i) {
177 double share = flex_share(flex[i], extra, fill_count, finite_grow);
178 double &main = stack_traits<Axis>::main(result.children[i]);
179 main += share;
180 main = std::max(0.0, main);
181 }
182}

184template <stack_axis Axis, std::ranges::input_range Range>
185 requires layout_element<std::ranges::range_value_t<Range>>
186auto measure_stack(Range const &children, proposal const &proposal, double spacing)
187 -> layout_result {
188 layout_result result;
189 std::vector<flex_policy> flex;
190 std::size_t child_count = 0;
191
192 for (auto &&child : children) {
193 auto child_size = clamp_size(child.measure(proposal), proposal);
194 flex_policy child_flex = layout_flex(child);
195
196 main_reference<Axis>(result.extent) += main_value<Axis>(child_size);
197 cross_reference<Axis>(result.extent) =
198 std::max(cross_value<Axis>(result.extent), cross_value<Axis>(child_size));
199 result.children.push_back(child_size);
200 flex.push_back(child_flex);
201 ++child_count;
202 }
203
204 if (child_count > 1) {
205 main_reference<Axis>(result.extent) += spacing * static_cast<double>(child_count - 1);
206 }
207
208 std::optional<double> max_main = stack_traits<Axis>::main_constraint(proposal).max;
209 double initial_main = main_value<Axis>(result.extent);
210
211 if (max_main && initial_main != *max_main) {
212 double extra = *max_main - initial_main;
213 distribute_flex<Axis>(result, flex, extra);
214 main_reference<Axis>(result.extent) = *max_main;
215 }
216
217 result.extent = clamp_size(result.extent, proposal);
218 return result;
219}

270inline auto arrange_hstack(std::span<size const> child_sizes, rect const &bounds,
271 layout_direction direction = layout_direction::left_to_right,
272 layout_alignment alignment = layout_alignment::start,
273 double spacing = 0.0) -> std::vector<rect> {
274 std::vector<rect> frames;
275 frames.reserve(child_sizes.size());
276
277 double x = direction == layout_direction::left_to_right ? bounds.origin.x
278 : bounds.origin.x + bounds.extent.width;
279
280 for (auto child_size : child_sizes) {
281 if (direction == layout_direction::right_to_left) {
282 x -= child_size.width;
283 }
284
285 double y = bounds.origin.y;
286 if (alignment == layout_alignment::center) {
287 y += (bounds.extent.height - child_size.height) * 0.5;
288 } else if (alignment == layout_alignment::end) {
289 y += bounds.extent.height - child_size.height;
290 } else if (alignment == layout_alignment::stretch) {
291 child_size.height = bounds.extent.height;
292 }
293
294 frames.push_back({{x, y}, child_size});
295
296 if (direction == layout_direction::left_to_right) {
297 x += child_size.width + spacing;
298 }
299 }
300
301 return frames;
302}

305inline auto arrange_vstack(std::span<size const> child_sizes, rect const &bounds,
306 layout_alignment alignment = layout_alignment::start,
307 double spacing = 0.0) -> std::vector<rect> {
308 std::vector<rect> frames;
309 frames.reserve(child_sizes.size());
310
311 double y = bounds.origin.y;
312 for (auto child_size : child_sizes) {
313 double x = bounds.origin.x;
314 if (alignment == layout_alignment::center) {
315 x += (bounds.extent.width - child_size.width) * 0.5;
316 } else if (alignment == layout_alignment::end) {
317 x += bounds.extent.width - child_size.width;
318 } else if (alignment == layout_alignment::stretch) {
319 child_size.width = bounds.extent.width;
320 }
321
322 frames.push_back({{x, y}, child_size});
323 y += child_size.height + spacing;
324 }
