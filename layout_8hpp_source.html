110inline auto flex_share(flex_policy policy, double extra, std::size_t fill_count, double finite_grow)
111 -> double {
112 if (extra > 0.0 && fill_count > 0) {
113 return policy.fill ? extra / static_cast<double>(fill_count) : 0.0;
114 }
115 if (extra < 0.0 && finite_grow > 0.0) {
116 return extra * policy.grow / finite_grow;
117 }
118 if (extra < 0.0 && fill_count > 0) {
119 return policy.fill ? extra / static_cast<double>(fill_count) : 0.0;
120 }
121 if (finite_grow > 0.0) {
122 return extra * policy.grow / finite_grow;
123 }
124 return 0.0;
125}

127template <stack_axis Axis>
128inline void distribute_flex(layout_result &result, std::span<flex_policy const> flex,
129 double extra) {
130 std::size_t fill_count = 0;
131 double finite_grow = 0.0;
132
133 for (auto policy : flex) {
134 if (policy.fill) {
135 ++fill_count;
136 } else {
137 finite_grow += policy.grow;
138 }
139 }
140
141 for (std::size_t i = 0; i < result.children.size(); ++i) {
142 double share = flex_share(flex[i], extra, fill_count, finite_grow);
143 double &main = stack_traits<Axis>::main(result.children[i]);
144 main += share;
145 main = std::max(0.0, main);
146 }
147}

149template <stack_axis Axis, std::ranges::input_range Range>
150 requires layout_element<std::ranges::range_value_t<Range>>
151auto measure_stack(Range const &children, proposal const &proposal, double spacing)
152 -> layout_result {
153 layout_result result;
154 std::vector<flex_policy> flex;
155 std::size_t child_count = 0;
156
157 for (auto &&child : children) {
158 auto child_size = clamp_size(child.measure(proposal), proposal);
159 flex_policy child_flex = layout_flex(child);
160
161 main_reference<Axis>(result.extent) += main_value<Axis>(child_size);
162 cross_reference<Axis>(result.extent) =
163 std::max(cross_value<Axis>(result.extent), cross_value<Axis>(child_size));
164 result.children.push_back(child_size);
165 flex.push_back(child_flex);
166 ++child_count;
167 }
168
169 if (child_count > 1) {
170 main_reference<Axis>(result.extent) += spacing * static_cast<double>(child_count - 1);
171 }
172
173 std::optional<double> max_main = stack_traits<Axis>::main_constraint(proposal).max;
174 double initial_main = main_value<Axis>(result.extent);
175
176 if (max_main && initial_main != *max_main) {
177 double extra = *max_main - initial_main;
178 distribute_flex<Axis>(result, flex, extra);
179 main_reference<Axis>(result.extent) = *max_main;
180 }
181
182 result.extent = clamp_size(result.extent, proposal);
183 return result;
184}

235inline auto arrange_hstack(std::span<size const> child_sizes, rect const &bounds,
236 layout_direction direction = layout_direction::left_to_right,
237 layout_alignment alignment = layout_alignment::start,
238 double spacing = 0.0) -> std::vector<rect> {
239 std::vector<rect> frames;
240 frames.reserve(child_sizes.size());
241
242 double x = direction == layout_direction::left_to_right ? bounds.origin.x
243 : bounds.origin.x + bounds.extent.width;
244
245 for (auto child_size : child_sizes) {
246 if (direction == layout_direction::right_to_left) {
247 x -= child_size.width;
248 }
249
250 double y = bounds.origin.y;
251 if (alignment == layout_alignment::center) {
252 y += (bounds.extent.height - child_size.height) * 0.5;
253 } else if (alignment == layout_alignment::end) {
254 y += bounds.extent.height - child_size.height;
255 } else if (alignment == layout_alignment::stretch) {
256 child_size.height = bounds.extent.height;
257 }
258
259 frames.push_back({{x, y}, child_size});
260
261 if (direction == layout_direction::left_to_right) {
262 x += child_size.width + spacing;
263 } else {
264 x -= spacing;
265 }
266 }
267
268 return frames;
269}

275 std::vector<rect> frames;
