95 auto measure(proposal const &proposal) const -> size {
96 if (kind == layout_kind::text) {
97 return measure_text(proposal);
98 }
99 if (kind == layout_kind::overlay) {
100 return clamp_size(measure_overlay(children, proposal).extent, proposal);

107 auto arrange(rect const &bounds) const -> layout_box {
108 layout_box box{to_string(kind), content, bounds, {}};
109
110 if (kind == layout_kind::text) {
111 box.bounds = {bounds.origin,
112 measure_text(proposal::bounded(bounds.extent.width, bounds.extent.height))};
113 return box;
114 }
115 if (children.empty()) {
116 box.bounds = {bounds.origin, {}};
117 return box;
118 }
119
120 auto child_frames = arrange_children(bounds);
121 box.children.reserve(children.size());
122 for (std::size_t i = 0; i < children.size(); ++i) {
123 box.children.push_back(children[i].arrange(child_frames[i]));
124 }
125 box.bounds = {bounds.origin, occupied_extent(bounds, box.children)};
126 return box;
127 }

140 auto measure_stack(proposal const &proposal, detail::stack_axis axis) const -> size {
141 auto content_proposal = inset_proposal(proposal, stack.padding);
142 layout_result result;
143 if (axis == detail::stack_axis::horizontal) {
144 result = measure_hstack(children, content_proposal, stack.spacing);
145 } else {
146 result = measure_vstack(children, content_proposal, stack.spacing);
147 }
148 return add_padding(clamp_size(result.extent, content_proposal), stack.padding);
149 }

162 static auto occupied_extent(rect const &bounds, std::span<layout_box const> boxes) -> size {
163 size result;
164 for (auto const &child : boxes) {
165 result.width = std::max(result.width,
166 child.bounds.origin.x + child.bounds.extent.width - bounds.origin.x);
167 result.height = std::max(result.height, child.bounds.origin.y + child.bounds.extent.height -
168 bounds.origin.y);
169 }
170 return result;
171 }
172
173 static auto inset_proposal(proposal value, edge_insets const &insets) -> proposal {
174 if (value.width.max) {
175 value.width.max = std::max(0.0, *value.width.max - insets.left - insets.right);
176 }
177 if (value.height.max) {
178 value.height.max = std::max(0.0, *value.height.max - insets.top - insets.bottom);
179 }
180 return value;
181 }
182
183 static auto add_padding(size value, edge_insets const &insets) -> size {
184 value.width += insets.left + insets.right;
185 value.height += insets.top + insets.bottom;
186 return value;
187 }

195inline auto materialize_layout(inspection_node const &node,
196 std::shared_ptr<text_measure_fn const> text_measure) -> layout_node {
197 auto kind = to_layout_kind(node.kind);
198 if (!kind) {
199 throw std::logic_error("unknown layout kind: " + node.kind);
200 }
201
202 layout_node result;
203 result.kind = *kind;
204 result.content = node.content;
205 if (result.kind == layout_kind::text) {
206 result.text_measure = text_measure;
207 result.policy.grow = 0.0;
208 }
209
210 result.children.reserve(node.children.size());
211 for (auto const &child : node.children) {
212 result.children.push_back(materialize_layout(child, text_measure));
213 }
214
215 return result;
216}
217
218inline auto materialize_layout(inspection_node const &node, text_measure_fn text_measure)
219 -> layout_node {
220 return materialize_layout(node, std::make_shared<text_measure_fn const>(std::move(text_measure)));
221}
