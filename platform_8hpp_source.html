114 struct null_text_measurer_impl : public text_measurer {
115 auto measure(std::string_view text, font_descriptor const &) const -> size override {
116 // Simple heuristic: 8 pixels per character, 16 pixels height
117 return {static_cast<double>(text.length()) * 8.0, 16.0};
118 }
119 auto measure_wrapped(std::string_view text, font_descriptor const &,

122 double lines = std::ceil(text.length() / chars_per_line);
