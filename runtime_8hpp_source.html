33struct state_key_hash {
34 auto operator()(state_key const &key) const noexcept -> std::size_t {
35 auto seed = std::hash<std::string>{}(key.path);
36 auto kind_hash = std::hash<std::type_index>{}(key.kind);
37 seed ^= kind_hash + 0x9e3779b9U + (seed << 6U) + (seed >> 2U);
38 seed ^= std::hash<std::string>{}(key.name) + 0x9e3779b9U + (seed << 6U) + (seed >> 2U);
39 return seed;
40 }

116 template <class T>
117 auto state_at(evaluation_context &context, std::string const &component_path,
118 std::type_index component_kind, std::string const &name, T initial) {
119 state_key key{component_path, component_kind, name};
120 auto &staged = context.staged;
121
122 auto it = staged.find(key);
123 if (it == staged.end()) {
124 auto committed = committed_state_slots_.find(key);
125 if (committed != committed_state_slots_.end()) {
126 staged.emplace(key, committed->second);
127 } else {
128 staged.emplace(key, std::move(initial));
129 }
130 it = staged.find(key);
131 }
132
133 if (it->second.type() != typeid(T)) {
134 throw std::logic_error("state slot type changed between evaluations");
135 }
136
137 return state_handle<T>(&std::any_cast<T &>(it->second));
138 }

169 template <class Range, class IdFn, class BodyFn>
170 auto evaluate(dynamic_list_expression<Range, IdFn, BodyFn> const &expression, std::string path,
171 evaluation_context &context) {
172 inspection_node node{"dynamic_list", {}, {}};
173 std::unordered_set<std::string> child_paths;
174
175 for (auto &&item : expression.items) {
176 auto raw_id = std::invoke(expression.id_fn, item);
177 using storage_id_t = typename id_storage<std::decay_t<decltype(raw_id)>>::type;
178 storage_id_t id{raw_id};
179 auto child_path = path + "/" + explicit_id_segment(id);
180
181 if (!child_paths.insert(child_path).second) {
182 throw std::logic_error("duplicate item id");
183 }
184
185 auto child = std::invoke(expression.body_fn, item);
186 node.children.push_back(evaluate(child, std::move(child_path), context));
187 }
188
189 return node;
190 }

209 template <class Tuple, std::size_t... I>
210 void append_children(inspection_node &node, Tuple const &children, std::string const &path,
211 evaluation_context &context, std::index_sequence<I...>) {
212 std::vector<std::string> child_paths;
213 std::unordered_set<std::string> sibling_paths;
214 child_paths.reserve(sizeof...(I));
215 (remember_child_path(child_paths, sibling_paths, std::get<I>(children), path, I), ...);
216 (node.children.push_back(
